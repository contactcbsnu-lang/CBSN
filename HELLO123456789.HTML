Google Sheets as Real-Time Backend Storage

Sync students, inquiries, notices, and marksheets directly with Google Sheets. Sends auto-email alerts on every new inquiry!

⚡ How to setup Google Sheets Backend in 2 minutes:
  1. Create a new spreadsheet at sheets.new (Name it CBSN-S Database).
  2. Click Extensions → Apps Script. Replace everything with the script below.
  3. Click Deploy → New deployment → Select type: Web app.
  4. Set Execute as: Me and Who has access: Anyone.
  5. Click Deploy, copy the Web App URL and paste it below!

Google Apps Script Code (Includes Auto-Email Alerts & Multi-Sheet separation):

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var data = JSON.parse(e.postData.contents);
    var action = data.action || 'SYNC';
    var sheetName = 'General_Log';

    if (action === 'INQUIRY_SUBMIT') sheetName = 'Inquiries';
    else if (action === 'STUDENT_REGISTRATION') sheetName = 'Students';
    else if (action === 'NOTICE_PUBLISH') sheetName = 'Notices';
    else if (action === 'ADMIT_CARD_ISSUE') sheetName = 'AdmitCards';
    else if (action === 'RESULT_PUBLISH') sheetName = 'Results';

    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      if (sheetName === 'Inquiries') {
        sheet.appendRow(['Timestamp', 'Ref ID', 'Parent Name', 'Student Name', 'Phone', 'Email', 'Grade', 'Category', 'Message']);
      } else {
        sheet.appendRow(['Timestamp', 'Action', 'Roll', 'Name', 'Class', 'Details JSON']);
      }
    }

    if (action === 'INQUIRY_SUBMIT') {
      var p = data.payload || {};
      sheet.appendRow([new Date(), p.refId || '', p.parentName || '', p.studentName || '', p.phone || '', p.email || '', p.grade || '', p.category || '', p.message || '']);

      // Optional: Auto-Email notification to school admin
      try {
        var adminEmail = Session.getActiveUser().getEmail();
        if (adminEmail) {
          MailApp.sendEmail({
            to: adminEmail,
            subject: 'New Admission Inquiry: ' + (p.studentName || 'Student'),
            body: 'A new inquiry has been submitted by ' + p.parentName + ' for ' + p.studentName + ' (' + p.grade + '). Phone: ' + p.phone + ', Email: ' + p.email + '. Message: ' + p.message
          });
        }
      } catch (err) {}
    } else {
      sheet.appendRow([new Date(), action, data.roll || '', data.name || '', data.class || '', JSON.stringify(data.payload || data)]);
    }

    return ContentService.createTextOutput(JSON.stringify({ status: 'success' })).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
